    @model_validator(mode="after")
    def _validate_from_or_user(self) -> "EmailOutputs":
        if not self.from_ and not self.smtp.user:
            err = "Either message.from_ or smtp.user must be set"
            raise ValueError(err)
        if not self.from_:
            self.from_ = self.smtp.user
        if not self.smtp.user:
            self.smtp.user = self.from_[1] if isinstance(self.from_, tuple) else self.from_
        return self

    def make_message(self, config: "Configuration") -> EmailMessage:
        output = self.output
        assert output is not None
        env = Environment(autoescape=True)
        mail_to: list[str | tuple[str | None, str]] = list(self.to)
        msg = EmailMessage(
            html=env.from_string(self.body or output.stem).render(config.parameters.model_dump(exclude_unset=True)),
            subject=env.from_string(self.subject or output.stem).render(config.parameters.model_dump(exclude_unset=True)),
            mail_from=self.from_,
            mail_to=mail_to,
            cc=self.cc,
            bcc=self.bcc,
        )
        msg.attach(filename=output.name, content_disposition="attachment", data=output.read_bytes())
        return msg

def _common_subject(subjects: list[str]) -> str:
    """Keep the prefix and suffix shared by every subject and replace the rest with ``*``."""
    common_subject = subjects[0]
    for subject in subjects[1:]:
        # Find common prefix
        prefix_len = 0
        for a, b in zip(common_subject, subject, strict=False):
            if a == b:
                prefix_len += 1
            else:
                break
        # Find common suffix, which may not overlap the prefix in either string
        max_suffix_len = min(len(common_subject), len(subject)) - prefix_len
        suffix_len = 0
        for a, b in zip(reversed(common_subject), reversed(subject), strict=False):
            if a == b and suffix_len < max_suffix_len:
                suffix_len += 1
            else:
                break
        # Build new common subject
        middle_len = max(len(common_subject), len(subject)) - prefix_len - suffix_len
        suffix = common_subject[len(common_subject) - suffix_len :]
        common_subject = common_subject[:prefix_len] + ("*" * middle_len) + suffix
    return common_subject


def email_postprocess(configs: list[Configuration], comma_or_common: Literal["common", "comma"] = "common") -> None:
    """A postprocess hook to email all output files after processing completes.
    NOTE: it is assumed that all EmailOutputs instances are configured identically,
    as happens in a multi run scenario.

    Args:
        configs (list[Configuration]): The list of nbprint configuration/s.

    """
    msgs: list[EmailMessage] = []
    smtp = None

    for config in configs:
        if isinstance(config.outputs, EmailOutputs):
            msgs.append(config.outputs.make_message(config=config))
            if smtp is None:
                smtp = config.outputs.smtp.model_dump(exclude_unset=True, exclude_none=True, exclude={"type_"})
                smtp["fail_silently"] = False
    if not msgs:
        return
    # Combine message contents, adjust email and body to be the common substring, and send
    msg = msgs[0]
    if len(msgs) > 1:
        # Concatenate bodies
        combined_body = "<br>".join([m.html or "" for m in msgs])
        msg.html = combined_body + "<br>"

        # Subject may be paramterized,
        # so find the overlapping parts of the strings
        # and replace the non-overlapping parts with "*"
        subjects = [m.subject or "" for m in msgs]

        if comma_or_common == "comma":
            msg.subject = ", ".join(subjects)
        else:
            msg.subject = _common_subject(subjects)

        # Attach all attachements
        for m in msgs[1:]:
            for attachment in m.attachments:
                msg.attachments.add(attachment)

    # Send the email
    response = msg.send(to=config.outputs.to, smtp=smtp)
    if not response.success:
        err = f"Failed to send email: {response.error}"
        raise RuntimeError(err)
